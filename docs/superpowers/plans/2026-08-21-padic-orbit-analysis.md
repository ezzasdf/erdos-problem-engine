```python
#!/usr/bin/env python3
"""p-Adic orbit-Cantor intersection analysis for the Erdős Ternary Conjecture.

Computes the intersection of the ×2 orbit in Z_3 with the Cantor set Σ_{3,2}
at each p-adic level k, tracking how the survival set shrinks.
"""

import math
from typing import List, Set, Dict, Tuple


def ternary_digits(n: int, k: int) -> List[int]:
    """Return the k least-significant ternary digits of n (LSB first)."""
    digits = []
    for _ in range(k):
        digits.append(n % 3)
        n //= 3
    return digits


def is_cantor(n: int, k: int) -> bool:
    """Check if n's k least-significant ternary digits are all in {0, 1}."""
    return all(d in (0, 1) for d in ternary_digits(n, k))


def cantor_residues(k: int) -> List[int]:
    """Return all r in [0, 3^k) with ternary digits in {0, 1}."""
    # 2^k residues, generated by iterating over all binary strings of length k
    residues = []
    for mask in range(1 << k):
        r = 0
        power = 1
        for i in range(k):
            if mask & (1 << i):
                r += power
            power *= 3
        residues.append(r)
    return sorted(residues)


def cantor_units(k: int) -> List[int]:
    """Return Cantor residues mod 3^k that are units (not divisible by 3)."""
    return [r for r in cantor_residues(k) if r % 3 != 0]
```

```python
def order_of_2_mod_3k(k: int) -> int:
    """Return the order of 2 mod 3^k, which is u_k = 2 * 3^(k-1)."""
    return 2 * (3 ** (k - 1))


def orbit_mod_3k(k: int) -> Dict[int, int]:
    """Return {2^n mod 3^k : n = 0, ..., u_k-1} mapped to their n values.

    Since 2 is a primitive root mod 3^k, this covers all of (Z/3^k Z)^×.
    Returns dict: {residue: smallest n with 2^n ≡ residue (mod 3^k)}.
    """
    uk = order_of_2_mod_3k(k)
    mod = 3 ** k
    result = {}
    power = 1  # 2^0 = 1
    for n in range(uk):
        if power not in result:
            result[power] = n
        power = (power * 2) % mod
    return result


def orbit_cantor_intersection(k: int) -> Set[int]:
    """Return {n mod u_k : 2^n mod 3^k ∈ C_k ∩ (Z/3^k Z)^×}.

    These are the n (mod u_k) for which 2^n is Cantor in the last k ternary digits.
    """
    mod = 3 ** k
    uk = order_of_2_mod_3k(k)
    cantor_set = set(cantor_units(k))
    result = set()
    power = 1  # 2^0 = 1
    for n in range(uk):
        if power in cantor_set:
            result.add(n)
        power = (power * 2) % mod
    return result
```

```python
def survival_set(k_max: int) -> Dict[int, Set[int]]:
    """Compute S_k = {n : 2^n mod 3^j ∈ C_j for all j ≤ k}, for k = 1..k_max.

    Returns dict: {k: set of n mod lcm(u_1,...,u_k) surviving all levels up to k}.

    Since u_k = 2*3^{k-1}, the lcm(u_1,...,u_k) = u_k (each divides the next).
    So S_k is a subset of {0, ..., u_k - 1}.
    """
    results = {}
    survivors = None  # will be set to orbit_cantor_intersection(1) initially

    for k in range(1, k_max + 1):
        intersection_k = orbit_cantor_intersection(k)
        uk = order_of_2_mod_3k(k)

        if survivors is None:
            # First level: S_1 = N_1
            survivors = intersection_k
        else:
            # S_k = S_{k-1} ∩ (intersection_k mod u_k)
            # Since u_{k-1} divides u_k, lift S_{k-1} to mod u_k and intersect
            lifted = set()
            for n in survivors:
                # n is in [0, u_{k-1}); it represents n + m*u_{k-1} for m = 0,1,2
                # But since the orbit is periodic with period u_k = 3*u_{k-1},
                # we need to check all lifts: n, n + u_{k-1}, n + 2*u_{k-1}
                u_prev = order_of_2_mod_3k(k - 1)
                for offset in range(3):  # 0, 1, 2
                    n_lifted = n + offset * u_prev
                    if n_lifted < uk and n_lifted in intersection_k:
                        lifted.add(n_lifted)
            survivors = lifted

        results[k] = set(survivors)

    return results


def print_survival_analysis(k_max: int = 15):
    """Print the survival set analysis table."""
    surv = survival_set(k_max)
    print(f"{'k':>3} {'u_k':>12} {'|C_k|':>8} {'|N_k|':>8} {'|S_k|':>8} {'ratio':>10} {'S_k content'}")
    print("-" * 80)
    for k in range(1, k_max + 1):
        uk = order_of_2_mod_3k(k)
        n_k = len(orbit_cantor_intersection(k))
        s_k = surv[k]
        n_k_count = len(orbit_cantor_intersection(k))
        s_k_count = len(s_k)
        ratio = s_k_count / n_k_count if n_k_count > 0 else 0
        # Show S_k content (mod 100 if too large)
        content = sorted(s_k)
        if len(content) > 10:
            content_str = f"{{{content[0]}, ..., {content[-1]}}} ({len(content)} elements)"
        else:
            content_str = str(content)
        print(f"{k:>3} {uk:>12} {2**k:>8} {n_k_count:>8} {s_k_count:>8} {ratio:>10.4f} {content_str}")
```

```python
def analyze_cantor_unit_structure(k_max: int = 10):
    """Analyze the algebraic structure of Cantor units mod 3^k."""
    print("\n=== Cantor Unit Structure ===")
    for k in range(1, k_max + 1):
        units = cantor_units(k)
        uk = order_of_2_mod_3k(k)
        mod = 3 ** k
        # Check if Cantor units form a subgroup under multiplication
        unit_set = set(units)
        is_subgroup = True
        for a in units:
            for b in units:
                if (a * b) % mod not in unit_set:
                    is_subgroup = False
                    break
            if not is_subgroup:
                break
        print(f"k={k}: |C_k units|={len(units)}, subgroup under ×? {is_subgroup}")


def analyze_orbit_pattern(k_max: int = 15):
    """Analyze how the orbit maps to Cantor residues."""
    print("\n=== Orbit-Cantor Mapping Pattern ===")
    for k in range(1, min(k_max + 1, 8)):
        uk = order_of_2_mod_3k(k)
        mod = 3 ** k
        intersection = orbit_cantor_intersection(k)
        # For each n in the intersection, show 2^n mod 3^k
        print(f"\nk={k} (u_k={uk}, mod={mod}):")
        for n in sorted(intersection):
            residue = pow(2, n, mod)
            digits = ternary_digits(residue, k)
            print(f"  n={n:>3}: 2^{n} mod {mod} = {residue:>3} = {''.join(str(d) for d in reversed(digits))}_3")


def analyze_2_to_n_ternary(k_max: int = 10):
    """Show the full ternary expansion of 2^n for n in {0, 2, 8}."""
    print("\n=== 2^n Ternary Expansions ===")
    for n in [0, 2, 8]:
        val = 2 ** n
        # Convert to ternary
        digits = []
        temp = val
        while temp > 0:
            digits.append(temp % 3)
            temp //= 3
        digits_str = ''.join(str(d) for d in reversed(digits))
        has_two = 2 in digits
        print(f"2^{n} = {val} = {digits_str}_3 (has digit 2: {has_two})")
```

```python
def cross_level_consistency(k_max: int = 15):
    """Check that S_{k+1} ∩ [0, u_k) = S_k (survival is monotone)."""
    print("\n=== Cross-Level Consistency ===")
    surv = survival_set(k_max)
    for k in range(1, k_max):
        s_k = surv[k]
        s_k1 = surv[k + 1]
        uk = order_of_2_mod_3k(k)
        # Lift S_k to [0, u_{k+1}) and check
        lifted = set()
        for n in s_k:
            for offset in range(3):
                lifted.add(n + offset * uk)
        consistent = lifted.issubset(s_k1)
        print(f"k={k}->{k+1}: S_k lifted ⊆ S_{{k+1}}? {consistent}  (|S_k|={len(s_k)}, |S_{{k+1}}|={len(s_k1)})")
```
